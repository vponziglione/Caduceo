<?php

$servername = "localhost";
$username = "root";
$password = "";
$dbname = "ospedale";


$conn = new mysqli($servername, $username, $password, $dbname);


if ($conn->connect_error) {
    die("Connessione fallita: " . $conn->connect_error);
}


$nome = $cognome = $cod_fiscale = $sesso = "";


if ($_SERVER["REQUEST_METHOD"] == "POST" && isset($_POST['id_paziente'])) {
    $id_paziente = $_POST['id_paziente'];

    $sql = "SELECT * FROM paziente WHERE id_paziente = ?";
    $stmt = $conn->prepare($sql);
    $stmt->bind_param("i", $id_paziente);
    $stmt->execute();
    $result = $stmt->get_result();

    if ($result->num_rows > 0) {
        $row = $result->fetch_assoc();
        $nome = $row['nome'];
        $cognome = $row['cognome'];
        $cod_fiscale = $row['cod_fiscale'];
        $sesso = $row['sesso'];
    } else {
        echo "Nessun paziente trovato con ID: $id_paziente";
    }

    $stmt->close();
}

$conn->close();
?>

            <h1 class="text-xl font-bold mb-6">CARTELLA CLINICA</h1>

                <label for="id_paziente" class="block text-gray-700 font-semibold mb-2">Inserisci ID Paziente:</label>
                <input 
                    type="text" 
                    id="id_paziente"

                    placeholder="Es. 123456"

                />
                <button

            <div class="grid grid-cols-2 gap-4 mb-6">
                <div class="bg-white p-4 shadow-md">
                    <p><strong>Paziente:</strong> <?php echo htmlspecialchars($nome . " " . $cognome); ?></p>
                    <p><strong>ID Paziente:</strong> <?php echo htmlspecialchars($id_paziente); ?></p>
                </div>
                <div class="bg-white p-4 shadow-md">
